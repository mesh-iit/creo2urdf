55inline std::map<ComponentId, std::string> resolveCadComponentNames(const std::map<ComponentId, std::string>& compID_cadName_map) {
56 // Count original CAD model names to identify models used by multiple occurrences.
57 std::map<std::string, size_t> cadName_occurrenceCount_map, candidateName_occurrenceCount_map;
58 for (const auto& componentModelEntry : compID_cadName_map) {
59 ++cadName_occurrenceCount_map[componentModelEntry.second];
60 }
61 // Build candidate names: preserve unique model names and append the full path
62 // to repeated ones. Reserve all candidates before resolving any collisions.
63 std::map<ComponentId, std::string> compID_cadOccurrenceName_map;
64 std::set<std::string> reservedNames_set;
65 for (const auto& componentModelEntry : compID_cadName_map) {
66 auto candidateCadName = componentModelEntry.second;
67 if (cadName_occurrenceCount_map.at(candidateCadName) > 1) {
68 candidateCadName += "_" + componentIdString(componentModelEntry.first);
69 }
70 compID_cadOccurrenceName_map.emplace(componentModelEntry.first, candidateCadName);
71 ++candidateName_occurrenceCount_map[candidateCadName];
72 reservedNames_set.insert(candidateCadName);
73 // Also avoid stealing any explicit path-qualified selector.
74 reservedNames_set.insert(componentModelEntry.second + "_" + componentIdString(componentModelEntry.first));
75 }
76 // Resolve collisions between generated candidates and existing CAD names,
77 // including collisions between two generated candidates from nested paths.
78 for (auto& componentCadNameEntry : compID_cadOccurrenceName_map) {
79 if (candidateName_occurrenceCount_map.at(componentCadNameEntry.second) == 1) {
80 continue;
81 }
82 // Preserve actual unique CAD names; disambiguate generated names only.
83 if (cadName_occurrenceCount_map.at(compID_cadName_map.at(componentCadNameEntry.first)) == 1) {
84 continue;
85 }
86 auto candidateCadName = componentCadNameEntry.second + "_occ_" + componentIdString(componentCadNameEntry.first);
87 // Insertion succeeds only for an unused name and reserves it immediately.
88 while (!reservedNames_set.insert(candidateCadName).second) {
89 candidateCadName += "_";
90 }
91 componentCadNameEntry.second = candidateCadName;
92 }
93 return compID_cadOccurrenceName_map;
94}

112inline std::map<ComponentId, std::string> resolveComponentURDFNames(const std::map<ComponentId, std::string>& compID_cadName_map, const std::map<ComponentId, std::string>& compID_URDFCompNames_map, const std::map<std::string, std::string>& renameKey_URDFName_map) {
113 // Prepare model multiplicities and default CAD occurrence names independently
114 // of the user's URDF naming choices.
115 std::map<std::string, size_t> cadName_occurrenceCount_map;
116 for (const auto& componentModelEntry : compID_cadName_map) {
117 ++cadName_occurrenceCount_map[componentModelEntry.second];
118 }
119 const auto compID_cadOccurrenceName_map = resolveCadComponentNames(compID_cadName_map);
120
121 // Validate explicit componentNames assignments before resolving rename keys.
122 for (const auto& componentAlias : compID_URDFCompNames_map) {
123 if (!compID_cadName_map.count(componentAlias.first) || componentAlias.second.empty()) {
124 throw std::runtime_error("Invalid componentNames entry: " + componentIdString(componentAlias.first));
125 }
126 }
127 // Map every supported textual selector to the occurrences it can identify.
128 // A set avoids counting the same occurrence twice when selector forms coincide.
129 std::map<std::string, std::set<ComponentId>> renameKey_compIDs_map;
130 for (const auto& componentModelEntry : compID_cadName_map) {
131 renameKey_compIDs_map[componentModelEntry.second + "_" + componentIdString(componentModelEntry.first)].insert(componentModelEntry.first);
132 if (cadName_occurrenceCount_map.at(componentModelEntry.second) == 1) {
133 renameKey_compIDs_map[componentModelEntry.second].insert(componentModelEntry.first);
134 }
135 renameKey_compIDs_map[compID_cadOccurrenceName_map.at(componentModelEntry.first)].insert(componentModelEntry.first);
136 }
137 // Reject ambiguous keys only when present in rename. This validation runs
138 // before alias precedence: an explicit alias does not excuse an ambiguous key.
139 for (const auto& renameSelectorEntry : renameKey_compIDs_map) {
140 if (renameSelectorEntry.second.size() > 1 && renameKey_URDFName_map.count(renameSelectorEntry.first)) {
141 throw std::runtime_error("Ambiguous occurrence rename key: " + renameSelectorEntry.first + "; use componentNames with explicit paths");
142 }
143 }
144 // Assign final URDF names per occurrence, tracking names already assigned
145 // so that explicit aliases and renames cannot produce duplicate links.
146 std::map<ComponentId, std::string> compID_URDFName_map;
147 std::set<std::string> assignedURDFNames_set;
148 for (const auto& componentModelEntry : compID_cadName_map) {
149 const auto componentAlias = compID_URDFCompNames_map.find(componentModelEntry.first);
150 const auto cadModelRename = renameKey_URDFName_map.find(componentModelEntry.second);
151 const auto pathQualifiedCadName = componentModelEntry.second + "_" + componentIdString(componentModelEntry.first);
152 const auto pathQualifiedRename = renameKey_URDFName_map.find(pathQualifiedCadName);
153 const auto cadOccurrenceRename = renameKey_URDFName_map.find(compID_cadOccurrenceName_map.at(componentModelEntry.first));
154 // Precedence: componentNames alias, path-qualified rename, resolved CAD
155 // occurrence rename, then unique model naming or the automatic default.
156 std::string urdfLinkName;
157 if (componentAlias != compID_URDFCompNames_map.end()) {
158 urdfLinkName = componentAlias->second;
159 } else if (pathQualifiedRename != renameKey_URDFName_map.end()) {
160 urdfLinkName = pathQualifiedRename->second;
161 } else if (cadOccurrenceRename != renameKey_URDFName_map.end()) {
162 urdfLinkName = cadOccurrenceRename->second;
163 } else if (cadName_occurrenceCount_map.at(componentModelEntry.second) == 1) {
164 urdfLinkName = cadModelRename == renameKey_URDFName_map.end() ? componentModelEntry.second : cadModelRename->second;
165 } else {
166 if (cadModelRename != renameKey_URDFName_map.end()) {
167 throw std::runtime_error("Ambiguous rename for " + componentModelEntry.second + "; use occurrence keys such as " + pathQualifiedCadName + " in rename, or componentNames");
168 }
169 urdfLinkName = compID_cadOccurrenceName_map.at(componentModelEntry.first);
170 }
171 // Reject invalid final names rather than silently altering user assignments.
172 if (urdfLinkName.empty() || !assignedURDFNames_set.insert(urdfLinkName).second) {
173 throw std::runtime_error("Duplicate or empty exported link name: " + urdfLinkName);
174 }
175 compID_URDFName_map.emplace(componentModelEntry.first, urdfLinkName);
176 }
177 return compID_URDFName_map;
178}
